# ==============================================================
#  mapas.txt  -  Archivo de definicion de mapas del motor
# ==============================================================
#
#  FORMATO DE UN BLOQUE DE MAPA:
#  --------------------------------
#  [MAPA]
#  nombre=Nombre del cuarto
#  spawn=X Y ANGULO
#  ANCHO ALTO
#  <filas del grid>
#  [FIN]
#
#  REGLAS DEL GRID:
#    - Cada caracter es una celda del mapa (digitos 0-9)
#    - 0 = celda vacia (el jugador puede caminar)
#    - 1-9 = pared con el tipo definido en config_paredes.cpp
#    - El ancho y alto deben coincidir exactamente con el grid
#    - Las lineas que empiezan con '#' son comentarios (ignoradas)
#
#  TABLA DE TIPOS DE PARED (definida en config_paredes.cpp):
#    1 = Piedra gris         (altura normal)
#    2 = Muro azul           (altura normal)
#    3 = Columna alta verde  (muy alta, estrecha)
#    4 = Ladrillo rojo       (altura normal)
#    5 = Barda baja          (apenas media pared)
#    6 = Madera              (altura normal)
#    7 = Torre alta celeste  (extremadamente alta)
#    8 = Muro violeta        (altura normal)
#    9 = Placa de metal      (altura normal)
#
#  ANGULO DE SPAWN en radianes:
#    0.0  = mirando al este (derecha)
#    1.57 = mirando al sur  (abajo)
#    3.14 = mirando al oeste (izquierda)
#    4.71 = mirando al norte (arriba)
#
# ==============================================================
