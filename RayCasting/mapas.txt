# ==============================================================
#  mapas.txt  -  Archivo de definicion de mapas del motor
# ==============================================================
#
#  FORMATO:
#  ─────────────────────────────────────────────────────────────
#  [MAPA]
#  nombre=Nombre del cuarto
#  spawn=X Y ANGULO
#  ANCHO ALTO
#  <filas del grid>
#  [FIN]
#
#  TABLA DE TIPOS DE PARED:
#  ─────────────────────────────────────────────────────────────
#  0 = Espacio vacio            (el jugador puede caminar)
#  1 = ParedPiedra              altura doble, textura '#' / '-'
#  2 = ParedLadrillo            altura doble, textura '+' / '-'
#  3 = ParedMadera              altura doble, textura '%' / '/'
#  4 = ParedAzul                altura doble, textura '#' / '-'
#  5 = ParedAcero               altura doble, textura '=' / '-'
#  6 = ParedAzulAlta            altura 3.5x,  textura '|' / ':'
#  7 = BardaLadrilloBaja        altura 0.5x,  textura '@' / 'O'
#  8 = Arbol                    tronco '|' color 96 + follaje '#' color 162
#  9 = Arco (sin colision)      solo dibuja marco superior, el jugador pasa
#
#  ANGULO DE SPAWN en radianes:
#    0.0  = mirando al este   1.57 = mirando al sur
#    3.14 = mirando al oeste  4.71 = mirando al norte
# ==============================================================
